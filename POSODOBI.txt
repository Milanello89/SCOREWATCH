#!/usr/bin/env bash
# ============================================================================
#  Rezultat - POSODOBITEV
#
#  Doda imena igralcev, statistiko dvoboja, graf poteka in deljenje rezultata.
#  Zamenja tri datoteke: index.html, sw.js in README.md.
#
#  Uporaba:
#    - to datoteko nalozi v repozitorij SCOREWATCH (Add file -> Upload files)
#    - odpri Codespace in v terminal natipkaj:
#          git pull && bash POSODOBI.txt
# ============================================================================

set -e

echo ""
echo "  Rezultat - posodobitev"
echo "  ----------------------"
echo ""

echo "  [1/3] Razpakiram nove datoteke ..."
sed -n '/^__PAYLOAD__$/,$p' "$0" | tail -n +2 | base64 -d | tar xz

if [ ! -f "index.html" ]; then
  echo "  NAPAKA: razpakiranje ni uspelo."
  exit 1
fi

echo "  [2/3] Odstranjujem namestitveno datoteko ..."
rm -f "$0" POSODOBI.txt POSODOBI_1.txt

echo "  [3/3] Posiljam na GitHub ..."
git add -A
git -c user.email="rezultat@local" -c user.name="Rezultat Setup" \
    commit -m "Imena igralcev, statistika, graf poteka in deljenje rezultata" >/dev/null || true

if git push 2>/dev/null; then
  echo "        objavljeno."
else
  echo ""
  echo "  Posiljanje ni uspelo. Poskusi rocno:   git push"
  echo ""
fi

echo ""
echo "  KONCANO."
echo ""
echo "  POMEMBNO - kako videti novo razlicico na telefonu:"
echo ""
echo "    Aplikacija je shranjena na napravi, zato se stara razlicica"
echo "    lahko se nekaj casa prikazuje. Naredi tako:"
echo ""
echo "      1. Pocakaj 2-3 minute, da GitHub Pages objavi spremembo."
echo "      2. Aplikacijo na zacetnem zaslonu popolnoma zapri"
echo "         (podrsaj navzgor in jo odstrani iz seznama odprtih)."
echo "      3. Odpri jo znova - nova razlicica se prenese sama."
echo ""
echo "    Ce se vedno vidis staro, jo odpri v Safariju na naslovu"
echo "    in enkrat osvezi stran."
echo ""

exit 0
